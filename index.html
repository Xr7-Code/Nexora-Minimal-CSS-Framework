install

three ways. pick one.

cdn (jsdelivr) 
<link rel="stylesheet" href="https://cdn.jsdelivr.net/gh/Xr7-Code/Nexora-Minimal-CSS-Framework@main/framework.css">
github pages 
<link rel="stylesheet" href="https://xr7-code.github.io/Nexora-Minimal-CSS-Framework/framework.css">
local 
<link rel="stylesheet" href="./framework.css">

optional js for mobile nav, active links and form feedback:

nexora.js 
<script src="./nexora.js" defer></script>

quick start

a full page in less than 30 lines.

index.html 
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>my site</title>
  <link rel="stylesheet" href="./framework.css">
</head>
<body>
  <a href="#main" class="skip-link">skip to content</a>

  <header class="navbar">
    <div class="container nav-inner">
      <a href="/" class="logo">brand<span>.</span></a>
      <button class="nav-toggle" aria-label="menu" aria-expanded="false">☰</button>
      <nav class="nav-links" aria-label="main">
        <a href="#">features</a>
        <a href="#">contact</a>
      </nav>
      <a href="#" class="btn btn-primary btn-small">get started</a>
    </div>
  </header>

  <main id="main">
    <section class="hero">
      <div class="container">
        <div class="hero-content">
          <h1>hello.</h1>
          <p class="lead">a minimal site in 10 lines.</p>
          <div class="hero-actions">
            <a href="#" class="btn btn-primary">start</a>
            <a href="#" class="btn btn-outline">docs</a>
          </div>
        </div>
      </div>
    </section>
  </main>

  <footer class="footer">
    <div class="container footer-inner">
      <p>© 2026</p>
    </div>
  </footer>

  <script src="./nexora.js" defer></script>
</body>
</html>
